# 風自然班表 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建置「風自然班表」— 讓 10～30 位員工月曆自排休假、送出後互看，主管編排最終班表，並即時標示代理人組別衝突（紅）與休假規則超額（橘）的多人排班 Web App，可列印與匯出。

**Architecture:** 靜態前端（vanilla HTML/CSS/JS ＋ Firebase compat SDK v10.12.0）部署於 Firebase Hosting site `fzshift-sk`；資料存 Firestore（集合 `shift_` 開頭）；主管帳號管理由本專案專屬 Cloud Function `shiftApi`（codebase `shift`）執行 Admin 操作，只允許 `@shift.local` 帳號。純邏輯抽成可測模組（Vitest），跨角色流程用 Playwright E2E（Emulator Suite）。

**Tech Stack:** Firebase (Auth Email/Password, Firestore, Hosting, Cloud Functions nodejs18)、Firebase compat SDK 10.12.0、Vanilla JS ES modules、Vitest、Playwright、Express (functions)、firebase-admin。

```js
export function ymOf(date) {
  const d = typeof date === "string" ? new Date(date + "T00:00:00") : date;
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
}
export function parseYm(ym) {
  const [y, m] = ym.split("-").map(Number);
  return { year: y, month: m };
}
export function daysInMonth(ym) {
  const { year, month } = parseYm(ym);
  const last = new Date(year, month, 0).getDate();
  return Array.from({ length: last }, (_, i) => `${ym}-${String(i + 1).padStart(2, "0")}`);
}
export function weekdayOf(dateStr) { return new Date(dateStr + "T00:00:00").getDay(); }
export function isValidDateStr(s) { return /^\d{4}-\d{2}-\d{2}$/.test(s) && !isNaN(new Date(s + "T00:00:00").getTime()); }
export function addMonthsYm(ym, delta) {
  const { year, month } = parseYm(ym);
  const d = new Date(year, month - 1 + delta, 1);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
}
export function formatMD(dateStr) { const { month } = parseYm(ymOf(dateStr)); return `${month}/${Number(dateStr.slice(8, 10))}`; }
export function weekdayZh(n) { return "日一二三四五六"[n]; }
```

```js
export function holidayDatesInMonth(ym) {
  const { year } = parseYm(ym);
  return (HOLIDAYS[String(year)] || []).filter((d) => d.startsWith(ym));

export function holidayDaysInMonth(ym) { return holidayDatesInMonth(ym).length; }
export function quotaForMonth(monthDoc, ym) {
  const o = monthDoc && monthDoc.quotaOverride;
  return typeof o === "number" && o >= 0 ? o : holidayDaysInMonth(ym);
}
```

```js
import { weekdayOf, weekdayZh } from "./dates.js";

export function countWeekday(dates, weekday) {
  return dates.filter((d) => weekdayOf(d) === weekday).length;
}
export function evaluateWarnings(dates, { rule, quota }) {
  const out = [];
  if (rule && rule.enabled) {
    const checks = [[5, rule.friMax, "週五"], [6, rule.satMax, "週六"], [0, rule.sunMax, "週日"]];
    for (const [wd, max, label] of checks) {
      const count = countWeekday(dates, wd);
      if (count > max) out.push({ type: "weekend", weekday: wd, count, max, message: `本月${label}休 ${count} 天，超過 ${max} 天` });
    }
  }
  if (typeof quota === "number" && dates.length > quota) {
    out.push({ type: "quota", total: dates.length, quota, message: `本月休 ${dates.length} 天，超過可休假日 ${quota} 天` });
  }
  return out;
}
```

```js
  const s = v == null ? "" : String(v);

export function withBom(csv) { return "\uFEFF" + csv; }

  const rows = [["姓名", ...leaves.length ? [] : []]];

  rows.length = 0; rows.push(header);

    rows.push([u.name, l ? (l.status === "submitted" ? "已送出" : "草稿") : "尚未送出", l ? l.dates.join(" ") : ""]);

export function finalScheduleMatrixCsv({ ym, users, shiftTypes, finalsDays }) {
  const { daysInMonth } = requireDatesHelper;
```

```js
    const token = (req.headers.authorization || "").split("Bearer ")[1];
    if (!token) return res.status(401).json({ error: "未提供 Token" });
    const decoded = await auth.verifyIdToken(token, true);
    if (decoded.shiftRole !== "manager") return res.status(403).json({ error: "無主管權限" });
    req.mgr = decoded;
    next();
  } catch { res.status(401).json({ error: "Token 無效" }); }
}
async function assertShiftUser(uid) {
  const rec = await auth.getUser(uid).catch(() => null);
  if (!rec || !isShiftEmail(rec.email)) { const e = new Error("非本系統帳號"); e.status = 403; throw e; }
  return rec;
}
async function setShiftRole(uid, shiftRole) {
  const u = await auth.getUser(uid);
  const claims = { ...(u.customClaims || {}), shiftRole };
  await auth.setCustomUserClaims(uid, claims);
}
```

```js
import { daysInMonth, weekdayOf } from "../lib/dates.js";

export function buildMonthGrid(ym) {
  const days = daysInMonth(ym);
  const firstWd = weekdayOf(days[0]);
  const cells = [];
  for (let i = 0; i < firstWd; i++) cells.push({ date: null, weekday: i, day: null, inMonth: false });
  for (const d of days) cells.push({ date: d, weekday: weekdayOf(d), day: Number(d.slice(8, 10)), inMonth: true });
  while (cells.length % 7 !== 0) cells.push({ date: null, weekday: cells.length % 7, day: null, inMonth: false });
  const weeks = [];
  for (let i = 0; i < cells.length; i += 7) weeks.push(cells.slice(i, i + 7));
  return weeks;
}
```

```js
async function load() {

(async () => {
```
